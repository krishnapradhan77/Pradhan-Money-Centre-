/* =========================================================
   JSONP REQUEST
========================================================= */

function jsonp(params) {

  return new Promise(
    function(resolve, reject) {

      const callbackName =
        "jsonp_" +
        Date.now() +
        "_" +
        Math.floor(
          Math.random() * 100000
        );


      const script =
        document.createElement("script");


      let finished = false;


      const cleanup =
        function() {

          try {

            delete window[callbackName];

          } catch (e) {

            window[callbackName] =
              undefined;

          }


          if (script.parentNode) {

            script.parentNode.removeChild(
              script
            );

          }

        };


      const timer =
        setTimeout(
          function() {

            if (finished) return;

            finished = true;

            cleanup();

            reject(
              new Error(
                "Request timeout"
              )
            );

          },
          15000
        );


      window[callbackName] =
        function(data) {

          if (finished) return;

          finished = true;

          clearTimeout(timer);

          cleanup();

          resolve(data);

        };


      const query =
        new URLSearchParams({

          ...params,

          callback:
            callbackName

        });


      script.src =
        API_URL +
        "?" +
        query.toString();


      script.async = true;


      script.onerror =
        function() {

          if (finished) return;

          finished = true;

          clearTimeout(timer);

          cleanup();

          reject(
            new Error(
              "Network error"
            )
          );

        };


      document
        .head
        .appendChild(script);

    }
  );

}


/* =========================================================
   LOAD COMPANIES

   Google Sheet ke saare tabs automatically
   dropdown mein load honge.

   Saved company automatically restore hogi.
========================================================= */

async function loadCompanies() {

  companySelect.disabled = true;

  companySelect.innerHTML =
    '<option value="">Loading Companies...</option>';


  let data = null;

  let lastError = null;

  for (let attempt = 1; attempt <= 3; attempt++) {

    try {

      data =
        await jsonp({
          action: "companies"
        });

      if (
        data &&
        data.success === true &&
        Array.isArray(data.companies)
      ) {

        break;

      }

      lastError =
        new Error(
          "Invalid companies response"
        );

    } catch (error) {

      lastError = error;

    }

    if (attempt < 3) {

      await new Promise(function(resolve) {

        setTimeout(
          resolve,
          700 * attempt
        );

      });

    }

  }

  try {

    if (
      !data ||
      data.success !== true ||
      !Array.isArray(data.companies)
    ) {

      throw (
        lastError ||
        new Error(
          "Invalid companies response"
        )
      );

    }


    const companies =
      data.companies
        .map(function(company) {

          return String(
            company || ""
          ).trim();

        })
        .filter(function(company) {

          return company !== "";

        })
        .filter(function(company, index, array) {

          return array.indexOf(company)
            === index;

        });


    companySelect.innerHTML =
      '<option value="">Select Company</option>';


    companies.forEach(
      function(company) {

        const option =
          document.createElement("option");


        option.value =
          company;


        option.textContent =
          company;


        companySelect.appendChild(
          option
        );

      }
    );


    if (companies.length === 0) {

      companySelect.innerHTML =
        '<option value="">No companies found</option>';

      return;
    }


    companySelect.disabled = false;


    let savedCompany = "";


    try {

      savedCompany =
        (
          localStorage.getItem(
            SAVED_COMPANY_KEY
          ) || ""
        ).trim();

    } catch (storageError) {

      console.warn(
        "Local storage unavailable:",
        storageError
      );

    }


    if (
      savedCompany &&
      companies.includes(savedCompany)
    ) {

      companySelect.value =
        savedCompany;


      currentCompany =
        savedCompany;

    }


    else if (savedCompany) {

      try {

        localStorage.removeItem(
          SAVED_COMPANY_KEY
        );

      } catch (storageError) {

        console.warn(
          "Unable to clear saved company:",
          storageError
        );

      }


      currentCompany =
        "";

    }


  } catch (error) {

    console.error(
      "Company loading error:",
      error
    );


    companySelect.disabled =
      true;


    companySelect.innerHTML =
      '<option value="">Unable to load companies</option>';


    showMessage(
      "Company list load nahi ho paayi. Please refresh the page.",
      "error"
    );

  }

}

    showPayoutMessage(
      "Registered email डालें।",
      "error"
    );

    payoutEmailInput.focus();

    return;

  }

  if (
    !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(
      email
    )
  ) {

    showPayoutMessage(
      "कृपया सही email address डालें।",
      "error"
    );

    payoutEmailInput.focus();

    return;

  }

  payoutSendOtpBtn.disabled = true;
  payoutResendBtn.disabled = true;

  const originalText =
    payoutSendOtpBtn.innerHTML;

  payoutSendOtpBtn.innerHTML =
    '<span class="loading-spinner"></span>Sending...';

  try {

    const data =
      await jsonp({
        action: "payoutSendOtp",
        email: email
      });

    if (
      data &&
      data.success === true
    ) {

      payoutEmailStep.style.display =
        "none";

      payoutOtpStep.style.display =
        "block";

      payoutResult.style.display =
        "none";

      payoutOtpInput.value = "";

      showPayoutMessage(
        "OTP आपके email पर भेज दिया गया है।",
        "success"
      );

      setTimeout(function() {
        payoutOtpInput.focus();
      }, 100);

      return;

    }

    showPayoutMessage(
      data && data.message
        ? data.message
        : "OTP भेजा नहीं जा सका।",
      "error"
    );

  } catch (error) {

    console.error(
      "Payout OTP error:",
      error
    );

    showPayoutMessage(
      "Network problem. Please try again.",
      "error"
    );

  } finally {

    payoutSendOtpBtn.disabled =
      false;

    payoutResendBtn.disabled =
      false;

    payoutSendOtpBtn.innerHTML =
      originalText;

  }

}


async function verifyPayoutOtp() {

  clearPayoutMessage();

  const email =

      .toLowerCase();

  const otp =
    payoutOtpInput.value

function renderPayoutProfile(data) {

  const name =
    String(data.name || "").trim();

  const mobile =
    String(data.mobile || "").trim();

  const email =
    String(data.email || "").trim();

  const payout =
    String(data.payout || "").trim();

  const photo =
    String(data.photo || "").trim();

  payoutName.textContent =
    name || "Customer";

  payoutMobile.textContent =
    mobile || "";

  payoutAmount.textContent =
    payout || "—";

  payoutEmailSmall.textContent =
    email;

  payoutAvatar.innerHTML = "";

  if (photo) {

    const img =
      document.createElement("img");

    img.alt =
      name || "Profile photo";

    img.src =
      photo;

    img.onload =
      function() {

        payoutAvatar.innerHTML = "";

        payoutAvatar.appendChild(img);

      };

    img.onerror =
      function() {

        renderPayoutInitials(name);

      };

  } else {

    renderPayoutInitials(name);

  }

}


function renderPayoutInitials(name) {

  const words =
    String(name || "C")
      .trim()
      .split(/\s+/)
      .filter(Boolean);

  let initials =
    words
      .slice(0, 2)
      .map(function(word) {
        return word
          .charAt(0)
          .toUpperCase();
      })
      .join("");

  if (!initials) {
    initials = "C";
  }

  payoutAvatar.textContent =
    initials;

}
